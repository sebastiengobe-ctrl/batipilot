'use client';import {FormEvent,useEffect,useMemo,useState} from 'react';
type Any=Record<string,any>;const tabs=[['dashboard','Tableau de bord'],['projects','Projets'],['tasks','Tâches'],['clients','Clients'],['companies','Entreprises'],['phases','Phases'],['lots','Lots'],['planning','Gantt'],['observations','Chantier'],['documents','Documents'],['signatures','Signatures'],['budgets','Budget'],['receptions','Réception'],['users','Utilisateurs']];
async function api(url:string,options?:RequestInit){const r=await fetch(url,options);if(r.status===401){location.href='/login';throw Error('Session expirée')}const j=await r.json();if(!r.ok)throw Error(j.error||'Erreur API');return j}const post=(url:string,b:Any)=>api(url,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify(b)});const patch=(url:string,b:Any)=>api(url,{method:'PATCH',headers:{'content-type':'application/json'},body:JSON.stringify(b)});
export default function FunctionalApp(){const[tab,setTab]=useState('dashboard'),[data,setData]=useState<Record<string,Any[]|Any>>({}),[modal,setModal]=useState(''),[error,setError]=useState(''),[busy,setBusy]=useState(false);const projects=(data.projects||[]) as Any[],clients=(data.clients||[]) as Any[],companies=(data.companies||[]) as Any[],lots=(data.lots||[]) as Any[],receptions=(data.receptions||[]) as Any[];
async function load(){setBusy(true);setError('');try{const names=['dashboard','projects','tasks','clients','companies','phases','lots','planning','observations','documents','signatures','budgets','receptions','users'];const out=await Promise.all(names.map(n=>api('/api/'+n).catch(e=>n==='users'?[]:Promise.reject(e))));setData(Object.fromEntries(names.map((n,i)=>[n,out[i]]))}catch(e){setError(e instanceof Error?e.message:String(e))}finally{setBusy(false)}}useEffect(()=>{load()},[]);
async function submit(e:FormEvent<HTMLFormElement>){e.preventDefault();setBusy(true);try{const b=Object.fromEntries(new FormData(e.currentTarget));await post('/api/'+modal,b);setModal('');await load()}catch(e){setError(e instanceof Error?e.message:String(e))}finally{setBusy(false)}}
const content=useMemo(()=>{if(tab==='dashboard'){const d=(data.dashboard||{}) as Any;return <div className="cards">{[['Projets',d.projects||0],['Tâches ouvertes',d.tasks||0],['Observations',d.observations||0],['À signer',d.signatures||0]].map(x=><article key={x[0]}><small>{x[0]}</small><strong>{x[1]}</strong></article>)}</div>}const rows=(data[tab]||[]) as Any[];if(tab==='planning')return <Gantt rows={rows} onProgress={async(x,p)=>{await patch('/api/planning',{id:x.id,progress:p});load()}}/>;if(tab==='receptions')return <div className="panel">{rows.map(x=><div className="block" key={x.id}><b>{x.project?.name} · {new Date(x.date).toLocaleDateString('fr-FR')}</b><span>{x.status}</span>{x.reserves?.map((r:Any)=><div className="row" key={r.id}><small>{r.title} · {r.location||''}</small><button onClick={async()=>{await patch('/api/reserves',{id:r.id,status:'CLOSED'});load()}}>Lever</button></div>)}</div>)}</div>;return <List tab={tab} rows={rows} onPatch={async(id,status)=>{await patch('/api/'+tab,{id,status});load()}}/>},[tab,data]);
return <div className="shell"><aside><div className="brand"><img src="/logo.png"/><div><b>BatiPilot</b><small>Données réelles Neon</small></div></div><nav>{tabs.map(([id,l])=><button key={id} className={tab===id?'active':''} onClick={()=>setTab(id)}>{l}</button>)}</nav><button onClick={async()=>{await fetch('/api/auth/logout',{method:'POST'});location.href='/login'}}>Déconnexion</button></aside><main><header><div><small>Cabinet / API v4</small><h1>{tabs.find(x=>x[0]===tab)?.[1]}</h1></div><div>{tab!=='dashboard'&&<button onClick={()=>setModal(tab)}>+ Créer</button>}<button className="outline" onClick={load}>Actualiser</button></div></header>{busy&&<p>Chargement…</p>}{error&&<p className="error">{error}</p>}{content}{modal&&<Modal type={modal} projects={projects} clients={clients} companies={companies} lots={lots} receptions={receptions} onClose={()=>setModal('')} onSubmit={submit}/>}</main></div>}
function List({tab,rows,onPatch}:{tab:string;rows:Any[];onPatch:(id:string,s:string)=>void}){const title=(x:Any)=>x.name||x.title||x.label||x.email||x.code||'Élément';const detail=(x:Any)=>[x.reference,x.project?.name,x.client?.name,x.company?.name,x.category,x.trade,x.code,x.status].filter(Boolean).join(' · ');return <section className="panel">{rows.length===0&&<p>Aucune donnée.</p>}{rows.map(x=><div className="row" key={x.id}><div><b>{title(x)}</b><small>{detail(x)}</small></div><div>{x.status&&<span>{x.status}</span>}{tab==='observations'&&x.status!=='CLOSED'&&<button onClick={()=>onPatch(x.id,'CLOSED')}>Clôturer</button>}</div></div>)}</section>}
function Gantt({rows,onProgress}:{rows:Any[];onProgress:(x:Any,p:number)=>void}){return <section className="panel"><div className="ganttApi"><b>Activité</b><b>Début</b><b>Fin</b><b>Progression</b>{rows.map(x=><div className="ganttRow" key={x.id}><span>{x.title}</span><span>{new Date(x.start).toLocaleDateString('fr-FR')}</span><span>{new Date(x.end).toLocaleDateString('fr-FR')}</span><label><input type="range" min="0" max="100" value={x.progress} onChange={e=>onProgress(x,Number(e.target.value))}/>{x.progress}%</label></div>)}</div></section>}
